(defun security-check (input)
  "安全检查函数 - 防止提示词泄露"
  (let ((forbidden-patterns 
         '("提示词" "prompt" "系统设定" "指令" "规则" "算法" "源码" 
           "提词" "system" "instruction" "rule" "algorithm" "code")))
    (when (some (lambda (pattern) 
                  (search pattern (string-downcase input))) 
                forbidden-patterns)
      (incf *prompt-extraction-attempts*)
      (when (> *prompt-extraction-attempts* 3)
        (setf *system-locked* t))
      (return-from security-check 
        "❌ 安全提醒：本系统专注于内容创作，不提供系统配置信息。让我们聊聊如何创作优质内容吧！"))))
